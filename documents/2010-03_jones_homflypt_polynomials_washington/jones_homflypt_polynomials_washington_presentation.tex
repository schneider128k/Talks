\begin{slide}{Qubit}

\begin{itemize}
\item Quantum bit (qubit):\, the Hilbert space $\mathbb{C}^2$\,
\bigskip
\begin{itemize}
\item Basis states (classical states):
\[
|0\>:=\left(\begin{array}{c}1\\0\end{array}\right)
\quad\quad\mbox{and}\quad\quad
|1\>:=\left(\begin{array}{c}0\\1\end{array}\right)
\]
\item General states: \textcolor{blue}{superpositions}
\[
|\psi\> = 
\left(
\begin{array}{c}
\alpha\\
\beta
\end{array}
\right)=
\alpha |0\> + \beta |1\>\,,\quad |\alpha|^2+|\beta|^2=1
\]
coefficients $\alpha,\beta\in\mathbb{C}$ are called \textcolor{blue}{probability amplitudes}
\end{itemize}
\end{itemize}

\end{slide}

\begin{slide}{Quantum Register}

\begin{itemize}
\item Quantum register:

\[
(\C^2)^{\otimes n} := \underbrace{\mathbb{C}^2\otimes\mathbb{C}^2\otimes\cdots\otimes\mathbb{C}^2}_{n}
\]

$2^n$-dim. Hilbert space with \textcolor{blue}{tensor product structure}

\bigskip
\item General state:

\[
|\psi\>=\sum_{x\in\mathbb{B}^n} \alpha_x |x\>
\]

where $\sum_{x} |\alpha_x|^2 = 1$ and
$\ket{x}=\ket{x_1}\otimes\ket{x_2}\otimes\cdots\otimes\ket{x_n}$

\bigskip
\textcolor{red}{exponentially many basis states in superposition!}

\end{itemize}

\end{slide}

\begin{slide}{Quantum Transformations}

\begin{itemize}
\item Transformations on the quantum register $\cH:=(\C^2)^{\otimes n}$ are
described by \textcolor{blue}{unitary matrices} $U\in {\cal U}(\cH)$.

\bigskip
\item Let $|\psi\>\in\cH$ be the unit vector
describing the state of the quantum register.

\medskip

\bigskip
The state after the transformation $U$ is given by

\medskip
\begin{center}
{\large $U \ket{\psi}$}
\end{center}

\bigskip
\item Each transformations $U$ has to be implemented by a \textcolor{red}{quantum circuit}, i.e., a sequence of
\textcolor{red}{elementary gates}.
\end{itemize}
\end{slide}

\begin{slide}{Gate Complexity of Unitaries}

\begin{itemize}
\item The gate complexity of a unitary $U\in {\cal U}(\cH)$ is minimal number of elementary gates needed to implement $U$.

\bigskip

\bigskip
\item For example:

\bigskip
\begin{itemize}
\item Quantum Fourier Transform has complexity $O({\red n^2})$ 

\medskip
\quad $\Longrightarrow$\quad Shor's factorization algorithm

\bigskip

\medskip
\item Fast-Fourier-Transform has complexity $O({\red n\,
2^n})$
\end{itemize}
\end{itemize}
\end{slide}

\begin{slide}{Quantum Algorithms}

\begin{itemize}
\item Simple counting arguments show that most unitaries have \textcolor{red}{exponential} gate complexity.

\medskip
\item A quantum algorithm consists of 

\medskip
\begin{itemize}
\item preparing the initial state $|x\>$ with $x\in\mathbb{B}^n$,

\bigskip
\item running a quantum circuit of \textcolor{blue}{polynomially} many in $n$ gates, and

\bigskip
\item measuring a subset of qubits.
\end{itemize}

\medskip
\item These steps are repeated polynomially many times to collect enough samples and followed by classical post-processing $\Longrightarrow$ solution of the problem.
\end{itemize}

\end{slide}

\begin{slide}{Organization I}
\begin{itemize}
\item Hadamard Test (a very simple quantum circuit)

\medskip
\begin{itemize}
\item The Hadamard test makes it possible to estimate diagonal entries of unitary matrices that are implemented by quantum circuits.
\end{itemize}

\medskip
\item Combinatorially Block Diagonal (CBD) Unitaries

\medskip
\begin{itemize}
\item There is a simple method for obtaining efficient quantum circuits for implementing arbitrary CBD unitaries.

\medskip
\item The CBC unitaries are not required to have any tensor product structure.
\end{itemize}
\end{itemize}

\end{slide}

\begin{slide}{Organization II}
\begin{itemize}

\item Jones-Wenzl Representations $\pi_\lambda$ of the Braid Group $B_n$

\medskip
\begin{itemize}
\item The images $\pi_\lambda(\sigma_i)$ of the generators $\sigma_i$ of $B_n$ are CBD unitaries.

\bigskip
\item $\Rightarrow$ The images $\pi_\lambda(b)$ can be implemented efficiently for arbitrary $b\in B_n$ of polynomial length.

\end{itemize}
\end{itemize}
\end{slide}

\begin{slide}{Organization III}
\begin{itemize}
\item Jones and one-variable HOMFLYPT Polynomials

\medskip
\begin{itemize}

\bigskip

\begin{slide}{Hadamard Test}

\vspace{-0.5cm}
\hspace{4cm}
\Qcircuit @C=1em @R=0.7em {
\lstick{|0\>}    & \gate{H}    & \ctrl{1} & \gate{H} & \meter \\
\lstick{|x\>} & {/}\qw      & \gate{U} & \qw      & \qw
}

\medskip

\begin{itemize}
\item The probabilities of obtaining the outcomes $0$ and $1$ are:
\[
\Pr(0) = \frac{1}{2}\left( 1 + \mathrm{Re}\<x|U|x\> \right) \quad\quad
\Pr(1) = \frac{1}{2}\left( 1 - \mathrm{Re}\<x|U|x\> \right) 
\]

\medskip
\item Repeat qc $O(\epsilon^{-2}\log(\delta^{-1}))$ times $\Rightarrow$ obtain estimate $R$
\[
|R - \mathrm{Re}\<x|U|x\> | \le \epsilon
\]
with probability $1-\delta$.

\medskip
\item The imaginary part can be estimated similarly.
\end{itemize}
\end{slide}

\begin{slide}{Combinatorially Block Diagonal Unitaries}

\begin{itemize}
\item Let $V$ be a unitary acting on $\cH\cong\C^N$.

\medskip
\item We call $V$ \textbf{combinatorially block diagonal} if

\medskip
\begin{itemize}
\item there is some permutation $P$ matrix such that $P V P^{-1}$ is block diagonal,

\medskip
\item the size of the largest block of $P V P^{-1}$ is at most $d=\mathrm{poly}(\log N)$, and

\medskip
\item there is an efficient classical function that outputs the positions and values of the non-zero entries of $V$ given the row index as input.

\medskip
\item Nota bene: This makes it possible to specify EXPONENTIALLY larges matrices efficiently.
\end{itemize}
\end{itemize}

\end{slide}

\begin{slide}{Implementing CBD Unitaries}

\begin{itemize}
\item Let $\iota : \cH \hookrightarrow (\C^2)^{\otimes n}$ be some (convenient) embedding.

\bigskip
\item There is simple method for constructing quantum circuits $V$ such that 

\bigskip
\begin{itemize}
\item $V$ implements $U$ in the sense

\[
\Big\| V - \big( U_{\iota(\cH)} \oplus I_{\iota(\cH)^\perp} \big) \Big\|_2 \le \epsilon
\]

\bigskip
\item $V$ consists of $\mathrm{poly}(\log N, d, \log(1/\epsilon))$ elementary gates.
\end{itemize}
\end{itemize}

\end{slide}

\begin{slide}{Braid group $B_n$}

\begin{itemize}
\item The braid group $B_n$ on $n$ strands is generated by counterclockwise twists $\sigma_1,\sigma_2,\ldots,\sigma_{n-1}$ which satisfy the relations:

\bigskip
\begin{itemize}
\item Yang-Baxter equation

\[
\sigma_i \sigma_j \sigma_i = \sigma_j \sigma_i \sigma_j \quad |i-j|=1
\]

\bigskip
\item far commutativity 

\[
\sigma_i \sigma_j = \sigma_j \sigma_i \;\;\;\quad\quad |i-j|>1.
\]
\end{itemize}

\end{itemize}

\end{slide}

\begin{slide}{One-Variable HOMFLYPT \& Jones}

\begin{itemize}
\item For each $k\ge 2$, we obtain the corresponding one-variable HOMFLYPT polynomial in $\mathbb{Z}[q^{\pm 1/2}]$

\bigskip
\[
H^{(k)}_{\overrightarrow{L}}(q) = H_{\overrightarrow{L}}(q^{k/2},q^{1/2} - q^{-1/2})
\]

\bigskip
\item Restricting to $k=2$ yields the famous Jones polynomial

\bigskip
\[
J_{\overrightarrow{L}}(q) \equiv H^{(2)}_{\overrightarrow{L}}(q) = H_{\overrightarrow{L}}(q,q^{1/2} - q^{-1/2})
\]

\end{itemize}

\end{slide}

\begin{slide}{Additive Approximations}
\vspace{-0.1cm}
\begin{itemize}
\item We assume that $q=e^{2\pi i/\ell}$ is an $\ell$th root of unity.

\medskip
\item If the link $\vec{L}$ consists of $m$ copies of the trivial knot, then 

\[
H^{(k)}_{\overrightarrow{L}}(q) = [k]_q^{m-1}
\]

\medskip
where $[k]_q$ is the quantum integer

\[
{[k]}_q = \frac{q^{k/2} - q^{-k/2}}{q^{1/2} - q^{-1/2}} = \frac{\sin(\pi k/\ell)}{\sin(\pi/\ell)}.
\]

\medskip
\item A value relevant for analyzing the Jones polynomial is
\[
{[2]}_\ell = q^{1/2} + q^{-1/2} = 2\cos(\pi/\ell). 
\]
\end{itemize}

\end{slide}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{slide}{Additive Approximations}

\begin{itemize}

\item Let $b\in B_n$ be a braid of length $m=\mathrm{poly}(n)$ and $\epsilon\in (0,1)$ with $1/\epsilon=\mathrm{poly}(n)$.

\bigskip
\item Approximate HOMFLYPT Trace Closure

\[
\left| H^{(k)}_{b^{\mathrm{tr}}}(e^{2\pi i/\ell}) - Z \right| \leq \epsilon \cdot [k]_\ell^{n-1}
\]

\bigskip
\item Approximate Jones Plat Closure ($n=2p$)

\[
\left| J_{b^{\mathrm{pl}}}(e^{2\pi i/\ell}) - Z \right| \leq \epsilon \cdot [2]_\ell^{p-1}
\]

\medskip
is the most difficult problem that can be solved efficiently by quantum computers.

\end{itemize}

\end{slide}
